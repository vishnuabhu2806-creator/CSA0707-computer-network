! ===== Router R2 (Cisco 2911) - EEE and Administration gateway, WAN DTE side =====
enable
configure terminal
hostname R2
no ip domain-lookup
!
interface GigabitEthernet0/0
 description LAN to SW-EEE (192.168.10.128/27)
 ip address 192.168.10.129 255.255.255.224
 no shutdown
!
interface GigabitEthernet0/1
 description LAN to SW-ADMIN (192.168.10.160/28)
 ip address 192.168.10.161 255.255.255.240
 no shutdown
!
interface Serial0/0/0
 description WAN to R1 (192.168.10.176/30)
 ip address 192.168.10.178 255.255.255.252
 no shutdown
!
router ospf 1
 router-id 2.2.2.2
 network 192.168.10.128 0.0.0.31 area 0
 network 192.168.10.160 0.0.0.15 area 0
 network 192.168.10.176 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
 passive-interface GigabitEthernet0/1
!
end
copy running-config startup-config
